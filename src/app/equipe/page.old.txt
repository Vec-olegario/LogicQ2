"use client";

import React, { useEffect, useState, useTransition } from "react";
import {
  Users,
  UserCheck,
  Clock,
  RotateCcw,
  Play,
  ShieldAlert,
  Sparkles,
  CheckCircle2,
  AlertCircle,
  Lock,
} from "lucide-react";
import { getSlotsDaEquipe, ocuparSlot, resetarEquipe } from "@/actions/slots";
import { iniciarTurno } from "@/actions/wms";

interface SlotData {
  id: string;
  papel: string;
  nomeDisplay: string | null;
  ocupado: boolean;
  ocupadoEm: Date | null;
  expiraEm: Date | null;
}

const PAPÉIS = [
  { id: "Líder", title: "Líder da Equipe", desc: "Supervisiona o turno e controla indicadores" },
  { id: "Recebimento", title: "Operador de Recebimento", desc: "Registra a chegada de cargas e notas físicas" },
  { id: "Estoque", title: "Operador de Estoque", desc: "Endereça paletes em ruas e prateleiras" },
  { id: "Picking", title: "Operador de Picking", desc: "Separa pedidos com validação por bipagem" },
  { id: "Expedição", title: "Operador de Expedição", desc: "Libera romaneios para docas de saída" },
];

export default function EquipePage() {
  const [equipeNome, setEquipeNome] = useState("Turma A - Grupo 1");
  const [equipeId, setEquipeId] = useState("equipe-demo-01");
  const [slots, setSlots] = useState<SlotData[]>([]);
  const [loading, setLoading] = useState(true);
  const [erroMsg, setErroMsg] = useState<string | null>(null);
  const [sucessoMsg, setSucessoMsg] = useState<string | null>(null);

  // Modal de ocupação
  const [slotParaOcupar, setSlotParaOcupar] = useState<SlotData | null>(null);
  const [nomeAluno, setNomeAluno] = useState("");
  
  // Modal de Reset
  const [resetModalOpen, setResetModalOpen] = useState(false);
  const [senhaAdmin, setSenhaAdmin] = useState("");

  const [isPending, startTransition] = useTransition();

  const carregarSlots = async () => {
    setLoading(true);
    setErroMsg(null);
    const res = await getSlotsDaEquipe(equipeId);
    if (res.sucesso) {
      setSlots(res.dados as SlotData[]);
    } else {
      setErroMsg(res.erro);
    }
    setLoading(false);
  };

  useEffect(() => {
    carregarSlots();
  }, [equipeId]);

  const handleOcupar = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!slotParaOcupar || !nomeAluno.trim()) return;

    setErroMsg(null);
    startTransition(async () => {
      const res = await ocuparSlot(slotParaOcupar.id, nomeAluno);
      if (res.sucesso) {
        setSucessoMsg(`Vaga de ${slotParaOcupar.papel} ocupada com sucesso por ${nomeAluno}!`);
        setSlotParaOcupar(null);
        setNomeAluno("");
        carregarSlots();
        setTimeout(() => setSucessoMsg(null), 4000);
      } else {
        setErroMsg(res.erro);
      }
    });
  };

  const handleIniciarTurno = async () => {
    setErroMsg(null);
    startTransition(async () => {
      const res = await iniciarTurno(equipeId);
      if (res.sucesso) {
        setSucessoMsg("Novo turno iniciado com sucesso! O placar de picking foi zerado.");
        setTimeout(() => setSucessoMsg(null), 4000);
      } else {
        setErroMsg(res.erro);
      }
    });
  };

  const handleResetEquipe = async (e: React.FormEvent) => {
    e.preventDefault();
    setErroMsg(null);
    startTransition(async () => {
      const res = await resetarEquipe(equipeId, senhaAdmin);
      if (res.sucesso) {
        setSucessoMsg(res.dados.mensagem);
        setResetModalOpen(false);
        setSenhaAdmin("");
        carregarSlots();
        setTimeout(() => setSucessoMsg(null), 4000);
      } else {
        setErroMsg(res.erro);
      }
    });
  };

  return (
    <div className="space-y-10">
      {/* ─── TOASTS ─── */}
      {sucessoMsg && (
        <div className="fixed top-20 right-6 z-50 bg-emerald-900 text-emerald-100 text-xs sm:text-sm font-semibold px-4 py-3 rounded-xl shadow-2xl border border-emerald-700 flex items-center gap-2 animate-in fade-in slide-in-from-top-2 duration-300">
          <CheckCircle2 className="w-4 h-4 text-emerald-400" />
          <span>{sucessoMsg}</span>
        </div>
      )}

      {erroMsg && (
        <div className="fixed top-20 right-6 z-50 bg-red-900 text-red-100 text-xs sm:text-sm font-semibold px-4 py-3 rounded-xl shadow-2xl border border-red-700 flex items-center gap-2 animate-in fade-in slide-in-from-top-2 duration-300">
          <AlertCircle className="w-4 h-4 text-red-400" />
          <span>{erroMsg}</span>
        </div>
      )}

      {/* ─── HEADER ─── */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 bg-white border border-zinc-200/80 rounded-3xl p-6 sm:p-8 shadow-[0_12px_40px_rgba(99,91,255,0.06)]">
        <div className="space-y-2">
          <div className="inline-flex items-center gap-2 bg-blue-50 border border-blue-200 px-3 py-1 rounded-full text-xs font-bold text-blue-700">
            <Users className="w-3.5 h-3.5" />
            <span>Sistema de Vagas Dinâmicas (Slots)</span>
          </div>
          <h1 className="text-3xl font-black text-zinc-900 tracking-tight">
            Gestão da Equipe: <span className="text-blue-600">{equipeNome}</span>
          </h1>
          <p className="text-sm text-zinc-600 max-w-xl">
            Escolha o seu papel no armazém para operar no celular sem precisar de login tradicional. O slot expira automaticamente após 4 horas.
          </p>
        </div>

        <div className="flex items-center gap-3">
          <button
            type="button"
            onClick={handleIniciarTurno}
            disabled={isPending}
            className="inline-flex items-center gap-2 bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 text-white font-bold px-5 py-3 rounded-2xl shadow-md transition-all hover:scale-105 cursor-pointer text-xs sm:text-sm disabled:opacity-50"
          >
            <Play className="w-4 h-4 fill-current" />
            <span>Iniciar Novo Turno</span>
          </button>

          <button
            type="button"
            onClick={() => setResetModalOpen(true)}
            className="inline-flex items-center gap-2 bg-zinc-100 hover:bg-zinc-200 text-zinc-700 font-bold px-4 py-3 rounded-2xl border border-zinc-300 transition-all hover:scale-105 cursor-pointer text-xs sm:text-sm"
          >
            <RotateCcw className="w-4 h-4 text-zinc-500" />
            <span>Reset Admin</span>
          </button>
        </div>
      </div>

      {/* ─── SLOTS GRID ─── */}
      <section className="space-y-4">
        <div className="flex items-center justify-between">
          <h2 className="text-xl font-bold text-zinc-900 tracking-tight flex items-center gap-2">
            <span>Vagas da Equipe</span>
            <span className="text-xs bg-zinc-100 text-zinc-600 px-2.5 py-0.5 rounded-full font-semibold">
              5 Papéis Disponíveis
            </span>
          </h2>

          <button
            onClick={carregarSlots}
            className="text-xs text-blue-600 hover:underline font-semibold flex items-center gap-1"
          >
            <Clock className="w-3.5 h-3.5" />
            <span>Atualizar status em tempo real</span>
          </button>
        </div>

        {loading ? (
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            {[1, 2, 3, 4, 5].map((i) => (
              <div key={i} className="h-44 bg-zinc-100 rounded-3xl animate-pulse" />
            ))}
          </div>
        ) : (
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            {PAPÉIS.map((papelInfo) => {
              const slotCadastrado = slots.find((s) => s.papel === papelInfo.id);
              const ocupado = slotCadastrado?.ocupado ?? false;
              const nomeAluno = slotCadastrado?.nomeDisplay;

              return (
                <div
                  key={papelInfo.id}
                  className={`bg-white border rounded-3xl p-6 shadow-sm flex flex-col justify-between transition-all duration-300 hover:shadow-md ${
                    ocupado
                      ? "border-emerald-200/80 bg-emerald-50/20"
                      : "border-zinc-200/80 hover:border-blue-300"
                  }`}
                >
                  <div className="space-y-4">
                    <div className="flex items-center justify-between">
                      <span className="text-xs font-bold uppercase tracking-wider text-zinc-400">
                        {papelInfo.id}
                      </span>
                      <span
                        className={`inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold ${
                          ocupado
                            ? "bg-emerald-100 text-emerald-800 border border-emerald-300"
                            : "bg-blue-50 text-blue-700 border border-blue-200"
                        }`}
                      >
                        {ocupado ? (
                          <>
                            <UserCheck className="w-3.5 h-3.5 text-emerald-600" />
                            <span>Ocupado</span>
                          </>
                        ) : (
                          <>
                            <Sparkles className="w-3.5 h-3.5 text-blue-600" />
                            <span>Vaga Livre</span>
                          </>
                        )}
                      </span>
                    </div>

                    <div>
                      <h3 className="text-lg font-bold text-zinc-900">{papelInfo.title}</h3>
                      <p className="text-xs text-zinc-500 mt-1 leading-relaxed">
                        {papelInfo.desc}
                      </p>
                    </div>

                    {ocupado && (
                      <div className="bg-white border border-emerald-200 rounded-2xl p-3 flex items-center justify-between">
                        <div>
                          <p className="text-[10px] uppercase font-bold text-zinc-400">Operador Atual</p>
                          <p className="text-sm font-bold text-emerald-900">{nomeAluno}</p>
                        </div>
                        <Clock className="w-4 h-4 text-emerald-600" />
                      </div>
                    )}
                  </div>

                  <div className="pt-6">
                    {ocupado ? (
                      <div className="text-center py-2 text-xs font-semibold text-zinc-400 bg-zinc-100 rounded-xl">
                        Vaga Ocupada (Expira em 4h)
                      </div>
                    ) : (
                      <button
                        type="button"
                        onClick={() =>
                          setSlotParaOcupar(
                            slotCadastrado || {
                              id: `temp-${papelInfo.id}`,
                              papel: papelInfo.id,
                              nomeDisplay: null,
                              ocupado: false,
                              ocupadoEm: null,
                              expiraEm: null,
                            }
                          )
                        }
                        className="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-3 rounded-2xl text-xs shadow-sm transition-all hover:scale-[1.02] cursor-pointer"
                      >
                        Ocupar esta Vaga
                      </button>
                    )}
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </section>

      {/* ─── MODAL OCUPAR SLOT ─── */}
      {slotParaOcupar && (
        <div className="fixed inset-0 z-50 bg-zinc-900/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white border border-zinc-200 rounded-3xl p-6 sm:p-8 max-w-md w-full shadow-2xl space-y-6 animate-in zoom-in-95 duration-200">
            <div className="space-y-1">
              <div className="text-xs font-bold text-blue-600 uppercase tracking-wider">
                Ocupação de Vaga
              </div>
              <h3 className="text-2xl font-bold text-zinc-900">
                Assumir Papel: {slotParaOcupar.papel}
              </h3>
              <p className="text-xs text-zinc-500">
                Digite seu nome ou apelido para ser identificado no painel da turma durante a aula prática.
              </p>
            </div>

            <form onSubmit={handleOcupar} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-zinc-700 uppercase mb-1">
                  Seu Nome de Exibição
                </label>
                <input
                  type="text"
                  required
                  value={nomeAluno}
                  onChange={(e) => setNomeAluno(e.target.value)}
                  placeholder="Ex: Vitor Hugo (Aluno 04)"
                  className="w-full bg-zinc-50 border border-zinc-200 rounded-2xl px-4 py-3 text-sm text-zinc-900 placeholder-zinc-400 focus:outline-none focus:ring-2 focus:ring-blue-600"
                />
              </div>

              <div className="flex gap-3 pt-2">
                <button
                  type="button"
                  onClick={() => setSlotParaOcupar(null)}
                  className="flex-1 bg-zinc-100 hover:bg-zinc-200 text-zinc-700 font-bold py-3 rounded-2xl text-xs cursor-pointer"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  disabled={isPending}
                  className="flex-1 bg-blue-600 hover:bg-blue-700 text-white font-bold py-3 rounded-2xl text-xs shadow-md cursor-pointer disabled:opacity-50"
                >
                  {isPending ? "Confirmando..." : "Confirmar Ocupação"}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ─── MODAL RESET ADMIN ─── */}
      {resetModalOpen && (
        <div className="fixed inset-0 z-50 bg-zinc-900/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white border border-zinc-200 rounded-3xl p-6 sm:p-8 max-w-md w-full shadow-2xl space-y-6 animate-in zoom-in-95 duration-200">
            <div className="flex items-center gap-3">
              <div className="w-10 h-10 rounded-2xl bg-amber-100 text-amber-700 flex items-center justify-center">
                <ShieldAlert className="w-5 h-5" />
              </div>
              <div>
                <h3 className="text-xl font-bold text-zinc-900">Reset de Equipe (Admin)</h3>
                <p className="text-xs text-zinc-500">Liberar todas as vagas e apagar histórico de turnos</p>
              </div>
            </div>

            <form onSubmit={handleResetEquipe} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-zinc-700 uppercase mb-1">
                  Senha do Professor / Administrador
                </label>
                <div className="relative">
                  <input
                    type="password"
                    required
                    value={senhaAdmin}
                    onChange={(e) => setSenhaAdmin(e.target.value)}
                    placeholder="Digite a senha admin"
                    className="w-full bg-zinc-50 border border-zinc-200 rounded-2xl pl-10 pr-4 py-3 text-sm text-zinc-900 placeholder-zinc-400 focus:outline-none focus:ring-2 focus:ring-blue-600"
                  />
                  <Lock className="w-4 h-4 text-zinc-400 absolute left-3.5 top-3.5" />
                </div>
              </div>

              <div className="flex gap-3 pt-2">
                <button
                  type="button"
                  onClick={() => setResetModalOpen(false)}
                  className="flex-1 bg-zinc-100 hover:bg-zinc-200 text-zinc-700 font-bold py-3 rounded-2xl text-xs cursor-pointer"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  disabled={isPending}
                  className="flex-1 bg-red-600 hover:bg-red-700 text-white font-bold py-3 rounded-2xl text-xs shadow-md cursor-pointer disabled:opacity-50"
                >
                  {isPending ? "Resetando..." : "Executar Reset"}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
